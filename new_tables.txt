CREATE TABLE permissions (
    online_id INTEGER UNSIGNED NOT NULL PRIMARY KEY UNIQUE,
    level INTEGER NOT NULL DEFAULT 0
);
CREATE TABLE restrictions (
    online_id INTEGER UNSIGNED NOT NULL PRIMARY KEY UNIQUE,
    flags INTEGER UNSIGNED NOT NULL DEFAULT 0,
    kart_id VARCHAR(120)
);
CREATE TABLE IF NOT EXISTS wordlist (
    word TEXT PRIMARY KEY COLLATE NOCASE
);
CREATE TABLE feature_messages (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    server_uid TEXT NOT NULL,
    created_at DATETIME NOT NULL DEFAULT (datetime('now')),
    player_name TEXT NOT NULL,
    message TEXT NOT NULL
);

CREATE TABLE text_reports (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    server_uid TEXT NOT NULL,
    created_at DATETIME NOT NULL DEFAULT (datetime('now')),
    player_name TEXT NOT NULL,
    message TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS command_usage_log (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    server_uid TEXT NOT NULL,
    command_name TEXT NOT NULL,
    source TEXT NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX IF NOT EXISTS idx_command_usage_server_time
    ON command_usage_log(server_uid, created_at);

CREATE INDEX IF NOT EXISTS idx_command_usage_command
    ON command_usage_log(command_name);
